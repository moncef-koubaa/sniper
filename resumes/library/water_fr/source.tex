\section{PROJETS ACADÉMIQUES}
\project{Traitement Électrochimique d'un Rejet Industriel par Électro-floculation}
\begin{bul}
\item Élimination des cations toxiques (Cd$^{2+}$, Zn$^{2+}$) d'un effluent simulé, mesurée par un taux d'abattement de \textbf{85 \% (Cd$^{2+}$) et 57 \% (Zn$^{2+}$) en 20 min}, par électro-coagulation/floculation à électrodes d'aluminium.
\item Quantification de la cinétique d'élimination par polarographie impulsionnelle et loi de Fick.
\end{bul}
\project{Traitement d'un Effluent par Coagulation-Floculation}
\begin{bul}
\item Optimisation du pH de coagulation (Al$_2$(SO$_4$)$_3$/acide alginique), mesurée par un volume de floc décanté maximal à \textbf{pH = 8}, en testant 5 valeurs de pH sur effluent réel.
\item Interprétation de l'efficacité via la spéciation de l'aluminium (Al$^{3+}$/Al(OH)$_3$/Al(OH)$_4^-$), par diagramme de Pourbaix.
\end{bul}
\project{Polymères à Impression Moléculaire (MIPs) en Applications Pharmaceutiques}
\begin{bul}
\item Synthèse et caractérisation d'un MIP sélectif au sulpiride par polymérisation radicalaire, avec extraction Soxhlet, suivi par spectrophotométrie UV-Vis et validation par cinétique de libération contrôlée (240 min).
\item Synthèse d'un MIP core-shell MOF/polymère pour la délivrance ciblée de doxorubicine, avec \textbf{85 \% de taux d'encapsulation}.
\end{bul}
